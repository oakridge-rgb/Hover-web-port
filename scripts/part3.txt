function startSong(id){
  const s=songById(id);
  const ev=buildEvents(s);
  const beat=60000/s.bpm,secLen=beat*4;
  const pSecCount=[]; // player notes per section (camera focus)
  for(let i=0;i<Math.ceil(s.len/secLen);i++){
    const a=i*secLen;let n=0;
    for(const x of s.p)if(x[0]>=a&&x[0]<a+secLen)n++;
    pSecCount.push(n);
  }
  play={id,song:s,t:-4*beat,paused:false,over:false,
    startAt:0,perf0:0,frozenT:0,
    aEv:ev.aEv,gEv:ev.gEv,ai:0,gi:0,
    lanesP:[[],[],[],[]],opp:[],pp:[0,0,0,0],ap:[0,0,0,0],oPtr:0,
    score:0,combo:0,maxCombo:0,misses:0,breaks:0,accSum:0,accN:0,hits:0,
    cnt:{'SICK!!':0,'GOOD':0,'BAD':0,'SHIT':0},
    health:50,beat:-4,sec:-1,focus:'dad',pSecCount,
    zoomW:1,zoomH:1,camX:0,
    poseB:{lane:-1,until:-1e9},poseD:{lane:-1,until:-1e9},
    pops:[],splash:[],rec:[{press:-1e9,conf:-1e9},{press:-1e9,conf:-1e9},{press:-1e9,conf:-1e9},{press:-1e9,conf:-1e9}],
    oConf:[-1e9,-1e9,-1e9,-1e9],holds:[],bopT:0,missPoseT:0,done:false,flashT:0};
  for(const n of s.p)play.lanesP[n[1]].push({t:n[0],lane:n[1],sus:n[2],end:n[0]+n[2],hit:false,missed:false,done:false,dropped:false,tickN:0});
  for(const n of s.o)play.opp.push({t:n[0],lane:n[1],sus:n[2],end:n[0]+n[2]});
  play.opp.sort((a,b)=>a.t-b.t);
  if(AC){
    play.startAt=AC.currentTime+4*beat/1000+0.12;
    instBus.gain.cancelScheduledValues(AC.currentTime);
    instBus.gain.setValueAtTime(0.0001,AC.currentTime);
    instBus.gain.linearRampToValueAtTime(0.9,AC.currentTime+0.4);
    vocP.gain.setValueAtTime(1,AC.currentTime);
  }else play.perf0=performance.now()+4*beat;
  stopMenuMusic();
  ST='play';pauseSel=0;
  if(opts.flash)play.flashT=tGlob+0.12;
}
function songNow(){
  if(!play)return 0;
  if(play.paused)return play.frozenT;
  if(AC)return (AC.currentTime-play.startAt)*1000;
  return performance.now()-play.perf0;
}
function skipTo(ms){
  if(!play)return;
  const d=ms-play.t;
  if(AC)play.startAt-=d/1000;else play.perf0-=d;
  for(const arr of play.lanesP)for(const n of arr){
    if(n.end+180<ms&&!n.hit){n.hit=true;n.silent=true;if(n.sus>0)n.done=true;}
  }
  play.ai=0;while(play.ai<play.aEv.length&&play.aEv[play.ai].t<ms)play.ai++;
  play.gi=0;while(play.gi<play.gEv.length&&play.gEv[play.gi].t<ms)play.gi++;
  play.beat=Math.floor(ms/(60000/play.song.bpm));
  play.sec=Math.floor(ms/play.song.secLen||ms/(60000/play.song.bpm*4));
  play.sec=Math.floor(ms/(60000/play.song.bpm*4));
  play.health=70;play.t=ms;
}
function pauseGame(){
  if(!play||play.paused||play.over)return;
  play.paused=true;
  if(AC){play.frozenT=songNow();AC.suspend().catch(()=>{});}
  else play.frozenT=songNow();
  sPause();pauseSel=0;
}
function resumeGame(){
  if(!play||!play.paused)return;
  play.paused=false;
  if(AC)AC.resume().catch(()=>{});
  sPause();
}
function exitToMenu(){
  play=null;
  if(AC){AC.resume().catch(()=>{});}
  stopMenuMusic();startMenuMusic();
  ST='menu';menuT=0;
}
function killPlayer(){
  if(!play||play.over)return;
  play.over=true;deadT=0;
  sDeath();
  if(AC){instBus.gain.setTargetAtTime(0.0001,AC.currentTime,0.05);}
  deathOn=true;if(AC)deathNext=AC.currentTime+1.0;
  ST='dead';
}
function endSong(){
  if(!play||play.over)return;
  play.over=true;
  const p=play,s=p.song;
  const acc=p.accN?p.accSum/p.accN*100:0;
  const breaks=p.breaks;
  let fc=p.accN>0?'CLEAR':'';
  if(p.accN>0){
    if(breaks===0)fc=(p.cnt.GOOD+p.cnt.BAD+p.cnt.SHIT===0)?'SFC':(p.cnt.BAD+p.cnt.SHIT===0)?'GFC':'FC';
    else if(breaks===1)fc='SDCB';
  }
  let grade='F';
  if(acc>=100&&p.misses===0&&p.cnt.BAD+p.cnt.SHIT===0)grade='S+';
  else if(acc>=95)grade='S';else if(acc>=90)grade='A';else if(acc>=80)grade='B';
  else if(acc>=70)grade='C';else if(acc>=60)grade='D';
  const sc={score:p.score,acc:Math.round(acc*100)/100,grade:grade};
  p.newBest=false;
  const old=best[p.id];
  if(!opts.bot){
    if(!old||sc.score>old.score){best[p.id]=sc;p.newBest=true;store('fhm_best',best);}
  }
  p.res={acc:acc,fc:fc,grade:grade};
  if(AC){instBus.gain.setTargetAtTime(0.0001,AC.currentTime,0.15);}
  sConfirm();startMenuMusic();
  ST='results';
}
// ------------------------------------------------------------- judgement ----
const JW={SICK:45,GOOD:90,BAD:135,SHIT:166,MISS:180};
const WT={'SICK!!':1,'GOOD':0.67,'BAD':0.34,'SHIT':0};
function heldCount(l){return (heldKey[l]?1:0)+(heldTouch[l]?1:0)}
function pressLane(l){
  if(!play||play.paused||play.over||ST!=='play')return;
  play.rec[l].press=play.t+110;
  if(opts.bot)return;
  let bestN=null,bestD=1e9;
  for(let i=play.ap[l];i<play.lanesP[l].length;i++){
    const n=play.lanesP[l][i];
    if(n.hit||n.missed)continue;
    const d=play.t-n.t;
    if(d<-220)break;
    if(d>220){if(i===play.ap[l])play.ap[l]++;continue;}
    const a=Math.abs(d);
    if(a<=JW.SHIT&&a<bestD){bestD=a;bestN=n;}
  }
  if(bestN)hitNote(bestN,play.t-bestN.t);
  else if(!opts.ghost)ghostMiss();
}
function judgeOf(a){return a<=JW.SICK?'SICK!!':a<=JW.GOOD?'GOOD':a<=JW.BAD?'BAD':'SHIT'}
function hitNote(n,diff){
  const p=play,r=judgeOf(Math.abs(diff));
  n.hit=true;p.hits++;p.cnt[r]++;p.accSum+=WT[r];p.accN++;
  p.combo++;p.maxCombo=Math.max(p.maxCombo,p.combo);
  p.score+={'SICK!!':350,'GOOD':200,'BAD':100,'SHIT':50}[r];
  p.health=clamp(p.health+{'SICK!!':2.3,'GOOD':1.2,'BAD':-0.3,'SHIT':-1.2}[r],0,100);
  p.rec[n.lane].conf=play.t+130;
  addPop(r,n.lane);
  if(r==='SICK!!')addSplash(n.lane);
  if(opts.hsound)sTick();
  if(n.sus>0){p.holds.push(n);n.tickN=0;}
  if(p.health<=0&&!opts.bot)killPlayer();
}
function doMiss(n,ghost){
  const p=play;
  p.misses++;p.breaks++;p.combo=0;
  p.health=clamp(p.health-(ghost?2:4.5),0,100);
  if(!ghost)p.accN++;
  sMiss();vocDuck();p.missPoseT=play.t+400;
  if(p.health<=0&&!opts.bot)killPlayer();
}
function ghostMiss(){doMiss(null,true)}
function addPop(r,lane){
  const p=play;
  p.pops.push({img:SPR.rate[r],x:940+rnd(-30,30),y:330,vx:rnd(-40,40),vy:-rnd(200,300),rot:rnd(-0.12,0.12),life:0.8,digits:p.combo});
}
function addSplash(lane){
  const p=play,x=lanePX(lane),y=strumY();
  for(let i=0;i<10;i++){
    const ang=rnd(-Math.PI*0.9,-Math.PI*0.1);
    p.splash.push({x:x,y:y,vx:Math.cos(ang)*rnd(120,420),vy:Math.sin(ang)*rnd(160,430),
      rot:rnd(0,6.3),vr:rnd(-8,8),col:NOTE_COL[lane],life:rnd(0.22,0.38)});
  }
}

  for(const r of hitUI){

function lanePX(l){return (opts.middle?640:1010)-112+l*112}
function laneOX(l){return (opts.middle?640:270)-112+l*112}
function strumY(){return opts.down?610:120}
function noteY(t){
  const v=0.45*play.song.speed*(H/720),sy=strumY();
  return opts.down? sy-(t-play.t)*v : sy+(t-play.t)*v;

function updatePlay(dt){
  const p=play;
  if(!p||p.paused||p.over)return;
  p.t=songNow();
  pump();
  const t=p.t;
  // botplay
  if(opts.bot){
    for(let l=0;l<4;l++){
      for(let i=p.ap[l];i<p.lanesP[l].length;i++){
        const n=p.lanesP[l][i];
        if(n.t>t)break;
        if(!n.hit){n.hit=true;p.hits++;p.cnt['SICK!!']++;p.accSum+=1;p.accN++;p.score+=350;
          p.combo++;p.maxCombo=Math.max(p.maxCombo,p.combo);
          p.rec[l].conf=t+130;
          if(n.sus>0){p.holds.push(n);n.tickN=0;}
          if(i===p.ap[l])p.ap[l]++;
        }
      }
    }
  }else{
    // misses
    for(let l=0;l<4;l++){
      for(let i=p.ap[l];i<p.lanesP[l].length;i++){
        const n=p.lanesP[l][i];
        if(n.t>t-JW.MISS)break;
        if(!n.hit&&!n.missed){n.missed=true;doMiss(n,false);}
        if(i===p.ap[l])p.ap[l]++;
      }
    }
  }
  // holds
  for(const h of p.holds){
    if(h.done)continue;
    const end=h.t+h.sus;
    if(t>=end){h.done=true;h.held=p.health;continue;}
    const held=opts.bot||heldCount(h.lane)>0;
    if(!held){h.dropped=true;h.done=true;continue;}
    while(t>=h.t+(h.tickN+1)*125){h.tickN++;p.score+=10;p.health=clamp(p.health+0.3,0,100);}
  }
  // opponent receptor flash
  while(p.oPtr<p.opp.length&&p.opp[p.oPtr].t<=t){
    const n=p.opp[p.oPtr];p.oConf[n.lane]=t+140;p.oPtr++;
  }
  // beat / section
  const bpm=p.song.bpm,beatMs=60000/bpm;
  const beat=Math.floor(t/beatMs);
  if(beat!==p.beat){
    p.beat=beat;p.bopT=t;
    if(opts.cam){
      const secHas=p.pSecCount[Math.floor(t/(beatMs*4))]>0||t<0;
      if(beat%4===0&&secHas)p.zoomW=1.045;
      p.zoomH=1.03;
    }
  }
  const sec=Math.floor(t/(beatMs*4));
  if(sec!==p.sec){p.sec=sec;p.focus=(p.pSecCount[sec]||0)>0?'bf':'dad';}
  p.zoomW=lerp(p.zoomW,1,dt*7);p.zoomH=lerp(p.zoomH,1,dt*9);
  p.camX=lerp(p.camX,p.focus==='bf'?-110:110,dt*3.5);
  // end
  if(t>p.song.len)endSong();
}
function pump(){
  const p=play;
  if(!AC)return;
  const ahead=p.t+700;
  while(p.ai<p.aEv.length&&p.aEv[p.ai].t<ahead){
    const e=p.aEv[p.ai++];
    if(e.t>p.t-1200)fireEV(e,p.startAt+e.t/1000);
  }
  while(p.gi<p.gEv.length&&p.gEv[p.gi].t<=p.t){
    const e=p.gEv[p.gi++];
    if(e.k==='pose'){
      const o=e.w==='b'?p.poseB:p.poseD;
      o.lane=e.l;o.until=p.t+380;
    }
  }
}

      const px=l*W/4+8,pw2=W/4-16,py=H-104;
      const held=heldCount(l)>0;
      c.globalAlpha=held?0.34:0.13;
      c.fillStyle='#fff';c.beginPath();c.roundRect(px,py,pw2,92,16);c.fill();
      c.globalAlpha=held?0.9:0.5;

      c.globalAlpha=1;
      txt('DFJK'[l],px+pw2/2,py+80,17,'#fff','center',false,'#000',3);
    }
  }
}
function drawNotes(){
  const p=play,c=CX,sy=strumY();
  // receptors
  for(let l=0;l<4;l++){
    const px=lanePX(l),ox=laneOX(l);
    const conf=p.rec[l].conf>p.t,press=heldCount(l)>0;
    let img=SPR.rec[l];
    if(conf)img=SPR.conf[l];else if(press)img=SPR.recP[l];
    c.drawImage(img,px-50,sy-50,100,100);
    const oconf=p.oConf[l]>p.t;
    c.globalAlpha=opts.middle?0.3:0.9;
    c.drawImage(oconf?SPR.conf[l]:SPR.rec[l],ox-50,sy-50,100,100);
    c.globalAlpha=1;
  }
  // opponent notes
  for(let i=p.oPtr;i<p.opp.length;i++){
    const n=p.opp[i];
    if(n.t>p.t+1400)break;
    const y=noteY(n.t);
    if(y<-160||y>H+160)continue;
    c.globalAlpha=opts.middle?0.25:0.85;
    c.drawImage(SPR.note[n.lane],laneOX(n.lane)-35,y-35,70,70);
    if(n.sus>0){
      const y2=noteY(n.t+n.sus);
      drawTrail(laneOX(n.lane),y,y2,n.lane,n.t,false,null);
    }
    c.globalAlpha=1;
  }
  // player notes
  for(let l=0;l<4;l++){
    for(let i=p.ap[l];i<p.lanesP[l].length;i++){
      const n=p.lanesP[l][i];
      if(n.t>p.t+1400)break;
      if(n.hit&&!n.sus)continue;
      const y=noteY(n.t);
      if(y<-160||y>H+160)continue;
      c.globalAlpha=n.missed?0.3:1;
      if(n.sus>0&&(!n.hit||!n.done)){
        const y2=noteY(n.t+n.sus);
        drawTrail(lanePX(l),y,y2,l,n.t,n.hit,n);
      }
      if(!n.hit)c.drawImage(SPR.note[l],lanePX(l)-35,y-35,70,70);
      c.globalAlpha=1;
    }
  }
}
function drawTrail(x,yHead,yTail,lane,tStart,holding,noteRef){
  const c=CX,sy=strumY();
  const up=!opts.down;
  let y1=yHead,y2=yTail;
  if(up){ // upscroll: future notes have larger y; trail spans below the head
    if(holding)y1=Math.max(y1,sy);
    if(y2<=y1)return;
  }else{
    if(holding)y1=Math.min(y1,sy);
    if(y2>=y1)return;
  }
  const dropped=noteRef?noteRef.dropped:false;
  const base=c.globalAlpha;
  c.globalAlpha=base*(dropped?0.2:0.75);
  c.fillStyle=NOTE_COL[lane];
  c.fillRect(x-11,Math.min(y1,y2),22,Math.abs(y2-y1));
  c.globalAlpha=base;
  c.drawImage(SPR.cap[lane],x-31,y2-31,62,62);
  c.globalAlpha=1;
}
function drawPops(dt){
  const p=play,c=CX;
  for(let i=p.pops.length-1;i>=0;i--){
    const o=p.pops[i];
    o.life-=dt;o.x+=o.vx*dt;o.y+=o.vy*dt;o.vy+=900*dt;
    if(o.life<=0){p.pops.splice(i,1);continue;}
    c.save();c.globalAlpha=clamp(o.life/0.3,0,1);
    c.translate(o.x,o.y);c.rotate(o.rot);
    c.drawImage(o.img,-170,-45,340,90);
    if(o.digits>0){
      const ds=String(o.digits).split('');
      let dx=-ds.length*18;
      for(const dch of ds){c.drawImage(SPR.dig[+dch],dx,-90,50,62);dx+=44;}
    }
    c.restore();
  }
  for(let i=p.splash.length-1;i>=0;i--){
    const s=p.splash[i];
    s.life-=dt;s.x+=s.vx*dt;s.y+=s.vy*dt;s.vy+=700*dt;s.rot+=s.vr*dt;
    if(s.life<=0){p.splash.splice(i,1);continue;}
    c.save();c.globalAlpha=clamp(s.life/0.25,0,1);
    c.translate(s.x,s.y);c.rotate(s.rot);
    c.fillStyle=s.col;c.fillRect(-7,-7,14,14);
    c.restore();
  }
}
function drawPlay(dt){
  const p=play,c=CX;
  // world
  c.save();
  c.translate(W/2,H/2);c.scale(p.zoomW,p.zoomW);c.translate(-W/2+p.camX,-H/2);
  const beatPulse=clamp(1-(p.t-p.bopT)/(60000/p.song.bpm),0,1);
  p.beatPulse=beatPulse;
  drawStage(beatPulse,p.focus);
  const t=p.t;
  const bp=(o)=>o.until>t?o.lane:-1;
  const singB=(o)=>o.until>t;
  drawDad(330,600,1.05,bp(p.poseD),singB(p.poseD),beatPulse*0.6);
  drawBF(950,600,1.05,bp(p.poseB),singB(p.poseB),p.missPoseT>t,beatPulse*0.6);
  c.restore();
  // notes + hud
  drawNotes();
  drawHUD(dt);
  drawPops(dt);
  // countdown
  if(p.t<0){
    const beatMs=60000/p.song.bpm;
    const idx=Math.floor((p.t+beatMs*4)/beatMs); // 0..3
    const labels=['3','2','1','GO!'];
    const li=clamp(idx,0,3);
    const frac=((p.t+beatMs*4)%beatMs)/beatMs;
    const sc=1+0.25*clamp(1-frac*3,0,1);
    c.save();c.translate(W/2,H/2-40);c.scale(sc,sc);
    txt(labels[li],0,0,120,li===3?'#7fd4ff':'#fff','center',true,'#000',14);
    c.restore();
  }
  if(p.flashT>tGlob&&opts.flash){c.fillStyle='rgba(255,255,255,0.5)';c.fillRect(0,0,W,H);}
  if(p.paused)drawPause();
}

let lastTs=0;
function loop(ts){
  requestAnimationFrame(loop);
  const dt=Math.min(0.05,(ts-lastTs)/1000||0.016);lastTs=ts;
  tGlob+=dt;frame++;
  hitUI.length=0;
  if(ST==='title'){introIdx=Math.floor(tGlob/1.9);}
  if(ST==='menu')menuT+=dt;
  if(ST==='play')updatePlay(dt);
  if(ST==='dead')deadT+=0;
  beginFrame();
  if(ST==='title')drawTitle(dt);
  else if(ST==='menu')drawMenu(dt);
  else if(ST==='options')drawOptions(dt);
  else if(ST==='play')drawPlay(dt);
  else if(ST==='dead')drawDead(dt);
  else if(ST==='results')drawResults(dt);
  menuTick();deathTick();
}
