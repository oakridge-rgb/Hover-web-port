// ------------------------------------------------------------ music build --
// (PROG, ILVL, CHARTS come from charts.js)
const NOTE_L=[0,3,7,12];  // lane -> scale offset
function buildEvents(song){
  const ev=[],gEv=[],P=PROG[song.id];
  const secLen=60000/song.bpm*4;
  song.secLen=secLen;
  const oct=t=>12*(((Math.floor(t/secLen)>>2)%2));
  for(const n of song.p){
    ev.push({t:n[0],k:'bf',m:P.vbf+NOTE_L[n[1]]+oct(n[0]),d:0.13+n[2]/1000});
    gEv.push({t:n[0],k:'pose',w:'b',l:n[1]});
  }
  for(const n of song.o){
    ev.push({t:n[0],k:'dad',m:P.vdad+NOTE_L[n[1]]+oct(n[0]),d:0.19+n[2]/1000});
    gEv.push({t:n[0],k:'pose',w:'d',l:n[1]});
  }
  instFor(song,ev);
  const beat=60000/song.bpm;
  ev.push({t:-4*beat,k:'cnt',f:880,d:0.12});
  ev.push({t:-3*beat,k:'cnt',f:880,d:0.12});
  ev.push({t:-2*beat,k:'cnt',f:880,d:0.12});
  ev.push({t:-1*beat,k:'cnt',f:1760,d:0.4});
  ev.sort((a,b)=>a.t-b.t);gEv.sort((a,b)=>a.t-b.t);
  return{aEv:ev,gEv:gEv};
}
function instFor(song,ev){
  const P=PROG[song.id],st=song.secLen/16,nSec=Math.ceil(song.len/song.secLen);
  const lvlArr=ILVL[song.id]||[];
  for(let s=0;s<nSec;s++){
    const lvl=lvlArr[Math.min(s,lvlArr.length-1)]||1;
    const a=s*song.secLen,root=P.roots[s%4],ch=P.chords[s%4];
    const K=(stp,k,x)=>ev.push({t:a+stp*st,k:k,x:x});
    if(lvl===5){ // outro: final hit in first outro section
      if(s===nSec-2){K(0,'crash');K(0,'bassend',{m:root});K(0,'chordend',{m:ch});}
      continue;
    }
    if(lvl===1){
      K(0,'kick');K(8,'kick');if(s%2)K(10,'kick');
      K(4,'snare');K(12,'snare');
      for(let i=0;i<16;i+=2)K(i,'hat');
      for(let i=0;i<16;i+=2)K(i,'bass',{m:root+((i===6||i===14)?12:0),d:st/1000*1.8});
      K(0,'pad',{m:ch,d:song.secLen/1000});
    }else if(lvl===2){
      if(song.id==='fresh'){ // halftime breakdown
        K(0,'kick');K(10,'kick');K(8,'snare');
        K(0,'hat');K(4,'hat');K(8,'hat');K(12,'hat');K(14,'open');
        K(0,'bass',{m:root,d:st/1000*2.5});K(6,'bass',{m:root,d:st/1000*2.5});
        K(8,'bass',{m:root+3,d:st/1000*2.5});K(14,'bass',{m:root,d:st/1000*2.5});
        K(0,'pad',{m:ch,d:song.secLen/1000});
      }else{
        K(0,'kick');K(7,'kick');K(8,'kick');
        K(4,'snare');K(12,'snare');
        for(let i=0;i<16;i+=2)K(i,'hat');K(14,'open');
        K(0,'bass',{m:root,d:st/1000*1.6});K(3,'bass',{m:root,d:st/1000*1.4});
        K(6,'bass',{m:root+7,d:st/1000*1.4});K(8,'bass',{m:root,d:st/1000*1.6});
        K(11,'bass',{m:root+7,d:st/1000*1.4});K(14,'bass',{m:root,d:st/1000*1.4});
        K(4,'stab',{m:ch,d:0.18});K(12,'stab',{m:ch,d:0.18});
      }
    }else if(lvl===3){
      K(0,'kick');K(4,'kick');K(8,'kick');K(12,'kick');if(s%2)K(7,'kick');
      K(4,'snare');K(12,'snare');
      for(let i=0;i<16;i+=2)K(i,'hat');K(12,'hat');K(13,'hat');K(14,'hat');K(15,'hat');K(14,'open');
      for(let i=0;i<16;i+=2)K(i,'bass',{m:root+(i===6?12:0),d:st/1000*1.5});
      K(15,'bass',{m:root+12,d:st/1000});
      K(4,'stab',{m:ch,d:0.16});K(12,'stab',{m:ch,d:0.16});
    }else if(lvl===4){
      K(0,'crash');
      K(0,'kick');K(4,'kick');K(7,'kick');K(8,'kick');K(12,'kick');
      K(4,'snare');K(12,'snare');
      for(let i=0;i<16;i++)K(i,'hat');K(14,'open');
      for(let i=0;i<16;i+=2)K(i,'bass',{m:root+(i===6?12:0),d:st/1000*1.3});
      K(0,'arp',{m:root+12});
      K(4,'stab',{m:ch,d:0.14});K(12,'stab',{m:ch,d:0.14});
    }
  }
}
function fireEV(e,when){
  const k=e.k;
  if(k==='kick')kick(when);
  else if(k==='snare')snare(when);
  else if(k==='hat')hat(when,false);
  else if(k==='open')hat(when,true);
  else if(k==='crash')crash(when);
  else if(k==='bass')bass(when,e.x.m,e.x.d);
  else if(k==='stab')stab(when,e.x.m,e.x.d);
  else if(k==='pad')pad(when,e.x.m,e.x.d);
  else if(k==='arp')arpEv(when,e.x);
  else if(k==='bassend')bass(when,e.x.m,1.2,1.2);
  else if(k==='chordend'){stab(when,e.x.m,1.4,1.3);stab(when,e.x.m.map(m=>m+12),1.4,0.7);}
  else if(k==='cnt')sfxTone(e.f,e.f,e.d,'square',0.18);
  else if(k==='bf')blipBF(when,e.m,e.d);
  else if(k==='dad')blipDad(when,e.m,e.d);
}
// ------------------------------------------------------- menu / death loops -
let menuOn=false,menuNext=0,deathOn=false,deathNext=0;
const MENU_ROOTS=[45,41,48,43],MENU_CH=[[57,60,64],[53,57,60],[55,60,64],[55,59,62]];
const MENU_BAR=2.4;
function schedMenuBar(w0){
  const bar=Math.round(w0/MENU_BAR);
  const root=MENU_ROOTS[bar%4],ch=MENU_CH[bar%4],st=MENU_BAR/16;
  kick(w0,0.5);kick(w0+8*st,0.5);hat(w0+4*st,false,0.5);hat(w0+12*st,false,0.5);
  bass(w0,root,MENU_BAR*0.9,0.5);
  pad(w0,ch,MENU_BAR*0.98);
  pluck(w0+6*st,ch[1]+12,0.5);pluck(w0+14*st,ch[2]+12,0.5);
}
function menuTick(){
  if(!AC||!menuOn)return;
  if(menuNext<AC.currentTime)menuNext=AC.currentTime+0.05;
  while(menuNext<AC.currentTime+1.6){schedMenuBar(menuNext);menuNext+=MENU_BAR;}
}
function startMenuMusic(){menuOn=true;deathOn=false;if(AC)menuNext=AC.currentTime+0.1;}
function stopMenuMusic(){menuOn=false;deathOn=false;}
function schedDeathBar(w0){
  const st=1.6/16;
  kick(w0,0.8);kick(w0+3*st,0.6);
  pad(w0,[45,48,52],1.55);
}
function deathTick(){
  if(!AC||!deathOn)return;
  if(deathNext<AC.currentTime)deathNext=AC.currentTime+0.05;
  while(deathNext<AC.currentTime+1.8){schedDeathBar(deathNext);deathNext+=1.6;}
}

const NOTE_COL=['#C24B99','#00FFFF','#12FA05','#F9393F'];
const ROT=[-Math.PI/2,Math.PI,0,Math.PI/2];
const SPR={};
function arrowShape(c,u){
  c.beginPath();
  c.moveTo(0,-u);c.lineTo(0.95*u,-0.05*u);c.lineTo(0.38*u,-0.05*u);
  c.lineTo(0.38*u,0.95*u);c.lineTo(-0.38*u,0.95*u);c.lineTo(-0.38*u,-0.05*u);
  c.lineTo(-0.95*u,-0.05*u);c.closePath();
}
function mkCanvas(w,h){const c=document.createElement('canvas');c.width=w;c.height=h;return c}
function buildSprites(){
  const S=128,u=44;
  SPR.note=[];SPR.rec=[];SPR.recP=[];SPR.conf=[];SPR.hold=[];SPR.cap=[];
  for(let d=0;d<4;d++){
    // note
    let cv=mkCanvas(S,S),c=cv.getContext('2d');
    c.translate(S/2,S/2);c.rotate(ROT[d]);
    arrowShape(c,u);c.lineWidth=17;c.strokeStyle='#17172b';c.lineJoin='round';c.stroke();
    arrowShape(c,u);c.fillStyle=NOTE_COL[d];c.fill();
    arrowShape(c,u*0.82);c.lineWidth=6;c.strokeStyle='#fff';c.stroke();
    SPR.note.push(cv);
    // receptor static
    cv=mkCanvas(S,S);c=cv.getContext('2d');
    c.translate(S/2,S/2);c.rotate(ROT[d]);
    arrowShape(c,u);c.fillStyle='#26263a';c.fill();
    c.lineWidth=7;c.strokeStyle='#8f8fae';c.stroke();
    SPR.rec.push(cv);
    // receptor pressed
    cv=mkCanvas(S,S);c=cv.getContext('2d');
    c.translate(S/2,S/2);c.rotate(ROT[d]);c.scale(0.88,0.88);
    arrowShape(c,u);c.fillStyle='#1b1b2c';c.fill();
    c.lineWidth=7;c.strokeStyle='#5c5c7c';c.stroke();
    SPR.recP.push(cv);
    // receptor confirm
    cv=mkCanvas(S,S);c=cv.getContext('2d');
    c.translate(S/2,S/2);c.rotate(ROT[d]);c.scale(1.06,1.06);
    c.shadowColor=NOTE_COL[d];c.shadowBlur=22;
    arrowShape(c,u);c.fillStyle=NOTE_COL[d];c.fill();c.shadowBlur=0;
    arrowShape(c,u*0.82);c.lineWidth=6;c.strokeStyle='#fff';c.stroke();
    SPR.conf.push(cv);
    // hold piece + cap (drawn vertical pointing down: used with rotation 0)
    cv=mkCanvas(64,64);c=cv.getContext('2d');
    c.fillStyle=NOTE_COL[d];c.globalAlpha=0.85;
    c.fillRect(18,0,28,64);c.globalAlpha=1;
    c.strokeStyle='#17172b';c.lineWidth=5;c.strokeRect(18,-2,28,68);
    SPR.hold.push(cv);
    cv=mkCanvas(S,S);c=cv.getContext('2d');
    c.translate(S/2,S/2);c.rotate(ROT[d]);c.scale(0.62,0.62);
    arrowShape(c,u);c.fillStyle=NOTE_COL[d];c.fill();
    c.lineWidth=12;c.strokeStyle='#17172b';c.stroke();
    SPR.cap.push(cv);
  }
  // rating labels
  SPR.rate={};
  const RDEF=[['SICK!!','#7fd4ff'],['GOOD','#7fd457'],['BAD','#ff6656'],['SHIT','#b09a56']];
  for(const[nm,col]of RDEF){
    const cv=mkCanvas(340,90),c=cv.getContext('2d');
    c.font='italic 900 56px "Arial Black",Arial,sans-serif';
    c.textAlign='center';c.textBaseline='middle';
    c.lineWidth=10;c.strokeStyle='#111';c.strokeText(nm,170,48);
    c.fillStyle=col;c.fillText(nm,170,48);
    SPR.rate[nm]=cv;
  }
  // digits
  SPR.dig=[];
  for(let i=0;i<10;i++){
    const cv=mkCanvas(72,90),c=cv.getContext('2d');
    c.font='italic 900 62px "Arial Black",Arial,sans-serif';
    c.textAlign='center';c.textBaseline='middle';
    c.lineWidth=9;c.strokeStyle='#111';c.strokeText(i,36,48);
    c.fillStyle='#fff';c.fillText(i,36,48);
    SPR.dig.push(cv);
  }
}
