   640→async function fateJetBoot(){
   641→  if (FATE_JET_STATE === 'ready') return FATE_JET;
   642→  if (FATE_JET_BOOTING) return FATE_JET_BOOTING;
   643→  FATE_JET_STATE = 'booting'; FATE_JET_ERR = '';
   644→  try { sessionStorage.fateBootResume = '1'; } catch(e){} /* auto-resume after recovery reloads */
   645→  fateJetPaintStatus();
   646→  fateJetStage('starting boot\u2026');
   647→  FATE_JET_BOOTING = (async () => {
   648→    try {
   649→      await fateJetSwStart();
   650→      if (!navigator.serviceWorker.controller) return null; /* page reloading */
   651→    } catch(e){
   652→      FATE_JET_STATE = 'sw-failed'; FATE_JET_ERR = String(e.message || e);
   653→      FATE_JET_BOOTING = null; fateJetPaintStatus(); fateJetStage('service worker failed \u2014 tap to retry', 'err'); return null;
   654→    }
   655→    let coreUrl = null;
   656→    try {
   657→      fateJetStage('loading jet files\u2026');
   658→      fateJetStage('loading jet.core.js\u2026');

   660→      const apiP = fateJetLoadJs('jet.api.js');       /* defines $jetController (needs $jet) */
   661→      fateJetStage('loading jet.api.js\u2026');
   662→      /* utils MUST load after api: it destructures globalThis.$jetController at eval time */

   667→        fateJetResolveUrl('jet.inject.js'),
   668→        fateJetResolveUrl('jet.wasm')
   669→      ]);

   677→      }

   684→        config: { jetPath: coreUrl, wasmPath: wasmUrl, injectPath: injectUrl, virtualWasmPath: 'jet.wasm.js', codec: FATE_JET_CODEC, prefix: prefix }
   685→      });
   686→      /* ---- jet handshake: retry loop with FRESH MessageChannel port each round
   687→              (verbatim port of the official portal's boot recipe — without this,
   688→              a missed SW message = infinite "booting") ---- */

   690→      const hsPromise = new Promise(r => { hsResolve = r; });
   691→      const __origResolve = FATE_JET.readyResolve;
   692→      FATE_JET.readyResolve = function(){ try { hsResolve(true); } catch(e){} return __origResolve.apply(FATE_JET, arguments); };

   694→      for (let i = 0; !handshook; i++){

   697→          throw new Error('the service worker never answered the jet handshake');

   701→        try { FATE_JET.setupMessagePort(); } catch(e){}

   703→      }
   704→      await FATE_JET.ready;
   705→      FATE_JET.guardServiceWorkerRevive = false;
   706→      FATE_JET_STATE = 'ready'; FATE_JET_ERR = '';

   713→      fateJetStage('ready', 'ok');
   714→      setTimeout(() => { if (FATE_JET_STATE === 'ready') fateJetStage(null); }, 2500);
   715→      fateJetPaintStatus();
   716→      fateStartJetWatch();

   722→      } else {
   723→        FATE_JET_STATE = 'controller-failed'; FATE_JET_ERR = String(e.message || e);
   724→      }

   726→    }
   727→  })();

   729→     handshake retries legitimately take a while — a stalled boot is silent) */

   732→      clearInterval(wd);
   733→      FATE_JET_STATE = 'controller-failed'; FATE_JET_ERR = 'no boot progress for 75s \u2014 network or device too slow';
   734→      FATE_JET_BOOTING = null; fateJetPaintStatus(); fateJetStage('stuck \u2014 tap to retry', 'err');
   735→    }

   737→  }, 5000);
   738→  return FATE_JET_BOOTING;

   741→/* ---- SW health watcher (official recipe): periodic sw_alive probe, fresh port on
   742→        controllerchange, pending-RPC cleanup on pagehide. Survives iPad tab freezing. ---- */
   743→let fateWatchBusy = null;
   744→async function fateSwWatch(){
   745→  if (fateWatchBusy) return fateWatchBusy;
   746→  fateWatchBusy = (async () => {
   747→    try {
   748→      if (!FATE_JET || !navigator.serviceWorker) return;
   749→      const c = navigator.serviceWorker;
   750→      if (c.controller && FATE_JET.serviceWorkerController !== c.controller){
   751→        FATE_JET.serviceWorkerController = c.controller;
   752→        try { FATE_JET.setupMessagePort(); } catch(e){}
   753→      }
   754→      /* light health check: probe failure alone never resets (churn destabilizes);
   755→         the frame-level warming page handles genuine dead-SW recovery */
   756→      const reg = await c.getRegistration().catch(() => null);
   757→      if (reg && !await fateSwAlive(reg, 1)){
   758→        console.warn('[FATE v8.2] sw_alive light probe failed \u2014 will re-check next tick');
   759→      }
   760→    } catch(e){} finally { fateWatchBusy = null; }
   761→  })();
   762→  return fateWatchBusy;

   765→  try {
   766→    const c = navigator.serviceWorker;
   767→    setInterval(fateSwWatch, 20000);
   768→    addEventListener('focus', fateSwWatch);
   769→    document.addEventListener('visibilitychange', () => {
   770→      try { if (!document.hidden) { try { sessionStorage.removeItem('fateWarmingN'); } catch(e){} fateSwWatch(); } } catch(e){}

   775→        try { FATE_JET.setupMessagePort(); } catch(e){}

   782→      FATE_JET.rpc.recieve = function(e){

   793→        }
   794→        inflight.clear();
   795→      });
   796→    }
   797→  } catch(e){ console.warn('[FATE v8.2] watcher setup failed:', e); }
