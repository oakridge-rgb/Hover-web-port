   246→    let ws;
   247→    try{ ws = new WebSocket(fateWispUrl(server)); }catch(e){ return resolve(null); }
   248→    ws.binaryType = 'arraybuffer';
   249→    const id = (Math.floor(4294967294 * Math.random()) + 1) >>> 0;
   250→    let gotEcho = false, t0 = 0;
   251→    const finish = v => {
   252→      clearTimeout(timer);
   253→      try{ ws.onmessage = ws.onerror = ws.onclose = null; ws.close(); }catch(e){}
   254→      resolve(v);
   255→    };
   256→    const timer = setTimeout(() => finish(null), timeoutMs || 4500);
   257→    ws.onerror = () => finish(null);
   258→    ws.onclose = () => { if(!gotEcho) finish(null); };

   261→      if(!(d instanceof ArrayBuffer) || d.byteLength < 5) return finish(null);

   263→      const type = dv.getUint8(0), sid = dv.getUint32(1, true);
   264→      if(gotEcho){ if(sid === id) finish(Math.round(performance.now() - t0)); return; }
   265→      if(sid === 0){

   506→    loading: false, closed: false, frame: pane.querySelector('.fb-frame'), watch: null
   507→  };
   508→  tab.frame.addEventListener('load', () => {
   509→    clearTimeout(tab.watch);
   510→    tab.loading = false;
   511→    tab.pane.classList.remove('loading');
   512→    if (tab.url && !tab.atHome){ clearPaneStage(tab); syncTabMeta(tab); }
   513→    updateChrome();
   514→  });
   515→  wireStart(pane, tab);
   516→  paintRecents(pane);
   517→  TABS.push(tab);
   518→  ACTIVE = tab;
   519→  updateChrome();
   520→  if (url) navigate(tab, url);
   521→  return tab;
   522→}
   523→function wireStart(pane, tab){
   524→  pane.querySelector('.fb-search').addEventListener('keydown', e => {
   525→    if (e.key === 'Enter') navigate(tab, e.target.value);
   526→  });
   527→  pane.addEventListener('click', e => {
   528→    const go = e.target.closest('[data-go]');
   529→    if (go){ navigate(tab, go.getAttribute('data-go')); return; }
   530→    const act = e.target.closest('[data-act]');
   531→    if (!act) return;
   532→    if (act.getAttribute('data-act') === 'go') navigate(tab, pane.querySelector('.fb-search').value);
   533→    if (act.getAttribute('data-act') === 'rescan'){
   534→      toast('rescanning wisp lanes\u2026');
   535→      BOOT.state = 'idle';
   536→      boot().then(() => toast(BOOT.state === 'ready' ? 'tunnel ready via ' + laneHost(FATE_LANE.server) : 'rescan failed \u2014 ' + (BOOT.err || ''))).catch(() => {});
   537→    }
   538→  });
   539→}
   540→function paintRecents(pane){
   541→  const row = pane.querySelector('.js-recents');
   542→  let h = [];
   543→  try { h = _lsGet('fate_browser_history') || []; } catch(e){}

   550→    return;

   554→    b.className = 'fb-recent-chip';
   555→    b.setAttribute('data-go', x.u);
   556→    b.innerHTML = '<span class="dot"></span><span></span>';
   557→    b.lastChild.textContent = hostOf(x.u);
   558→    row.appendChild(b);
   559→  });
   560→}
   561→function activate(tab){
   562→  if (!tab || tab.closed) return;
   563→  ACTIVE = tab;
   564→  TABS.forEach(t => t.pane.classList.toggle('active', t === tab));
   565→  updateChrome();
   566→}
   567→function closeTab(tab){
   568→  if (!tab || tab.closed) return;
   569→  tab.closed = true;
   570→  clearTimeout(tab.watch);
   571→  const i = TABS.indexOf(tab);
   572→  TABS.splice(i, 1);
   573→  tab.pane.remove();
   574→  if (!TABS.length){ newTab(); return; }
   575→  if (ACTIVE === tab) activate(TABS[Math.max(0, i - 1)]);
   576→  updateChrome();
   577→}
   578→function navigate(tab, input){
   579→  if (!tab || tab.closed) return;
   580→  const url = parseInput(input);
   581→  if (!url){ toast('type a URL or search first'); return; }
   582→  tab.url = url; tab.atHome = false; tab.title = hostOf(url);
   583→  tab.pane.classList.remove('at-home');
   584→  setPaneStage(tab, 'connecting', 'loading ' + hostOf(url) + '\u2026',
   585→    BOOT.state === 'ready' ? '' : 'waiting for the wisp tunnel to start\u2026');
   586→  recordHistory(url);
   587→  updateChrome();
   588→  boot().then(() => {
   589→    if (tab.closed) return;
   590→    if (BOOT.state !== 'ready' || !FATE_LANE) return setPaneStage(tab, 'error', BOOT.err || 'tunnel failed');
   591→    setPaneStage(tab, 'connecting', 'loading ' + hostOf(url) + '\u2026', 'through ' + laneHost(FATE_LANE.server));
   592→    tab.loading = true;
   593→    tab.pane.classList.add('loading');
   594→    tab.frame.src = prox(url);
   595→    clearTimeout(tab.watch);
   596→    tab.watch = setTimeout(() => {
   597→      if (tab.loading) setPaneStage(tab, 'error', hostOf(url) + ' never answered through the tunnel',
   598→        'the lane may be too weak for this site \u2014 try rescan, or pick another lane in the scanner');
   599→    }, 45000);
   600→  }).catch(() => {
   601→    if (!tab.closed) setPaneStage(tab, 'error', BOOT.err || 'tunnel failed', 'no live wisp lane could start the tunnel');
   602→  });
   603→}
   604→function reloadTab(tab){
   605→  if (!tab) return;
   606→  if (tab.atHome || !tab.url){ toast('on the start page \u2014 nothing to reload'); return; }
   607→  try { tab.frame.contentWindow.location.reload(); }
   608→  catch(e){ tab.frame.src = prox(tab.url); }
   609→}
   610→function goHome(tab){
   611→  if (!tab) return;
   612→  tab.atHome = true;
   613→  tab.pane.classList.add('at-home');
   614→  clearPaneStage(tab);
   615→  updateChrome();
   616→}
   617→function setPaneStage(tab, kind, msg, note){
   618→  const p = tab.pane;
   619→  p.classList.toggle('err', kind === 'error');
   620→  p.classList.toggle('connecting', !!kind);
   621→  p.querySelector('.js-stage').textContent = msg || '';
   622→  p.querySelector('.js-stagenote').textContent = note || (kind === 'error' ? 'tap retry to boot the tunnel again' : '');
   623→}
   624→function clearPaneStage(tab){
   625→  tab.pane.classList.remove('connecting', 'err');
   626→}
   627→function syncTabMeta(tab){
   628→  if (!tab || tab.closed || tab.atHome || !tab.url) return;
   629→  try {
   630→    const w = tab.frame.contentWindow;
   631→    const real = decodeProxied(w.location);
   632→    if (real && /^https?:$/.test(real.protocol)) tab.url = real.href;
   633→    if (w.document && w.document.title) tab.title = String(w.document.title).slice(0, 80);
   634→  } catch(e){}
   635→  updateChrome();
   636→}
   637→function paintTabs(){
   638→  const el = $('fbTabs');
   639→  el.innerHTML = '';
   640→  TABS.forEach(t => {
   641→    const d = document.createElement('div');
   642→    d.className = 'fb-tab' + (t === ACTIVE ? ' active' : '');
   643→    const h = hueOf(t.atHome ? 'fate-home' : hostOf(t.url || 'fate'));
   644→    const ic = t.loading
   645→      ? '<span class="fb-tab-ic loading"></span>'
   646→      : '<span class="fb-tab-ic" style="background:hsla(' + h + ',45%,16%,.95);color:hsl(' + h + ',85%,70%)">' + (t.atHome ? 'F' : (hostOf(t.url || 'F')[0] || 'F').toUpperCase()) + '</span>';
   647→    d.innerHTML = ic + '<span class="fb-tab-title"></span><span class="fb-tab-x" title="Close tab">\u00d7</span>';
   648→    d.querySelector('.fb-tab-title').textContent = t.atHome ? 'New tab' : (t.title || hostOf(t.url || ''));
   649→    d.addEventListener('click', () => activate(t));
   650→    d.addEventListener('auxclick', e => { if (e.button === 1) closeTab(t); });
   651→    d.querySelector('.fb-tab-x').addEventListener('click', e => { e.stopPropagation(); closeTab(t); });
   652→    el.appendChild(d);
   653→  });
   654→  const act = el.querySelector('.fb-tab.active');
   655→  if (act && act.scrollIntoView) try { act.scrollIntoView({ block: 'nearest', inline: 'nearest' }); } catch(e){}
   656→}
   657→function updateChrome(){
   658→  paintTabs();
   659→  const t = ACTIVE;
   660→  const off = !t || t.atHome || !t.url;
   661→  $('fbBackBtn').disabled = off;
   662→  $('fbFwdBtn').disabled = off;
   663→  $('fbRelBtn').disabled = off;
   664→  $('fbPopBtn').disabled = off;
   665→  const a = $('fbAddr');
   666→  if (document.activeElement !== a) a.value = off ? '' : (t.url || '');
   667→  document.title = t && !t.atHome && t.title ? (t.title + ' \u00b7 FATE Browser') : 'FATE Browser';
   668→}
