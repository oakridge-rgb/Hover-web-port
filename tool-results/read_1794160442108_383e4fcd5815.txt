   306→  var bytes = b64Bytes(b64);
   307→  if (bytes.length > 2 && bytes[0] === 0x1f && bytes[1] === 0x8b){
   308→    if (typeof DecompressionStream !== 'function'){
   309→      return Promise.reject(new Error('browser too old - needs DecompressionStream (Chrome 80+ / Safari 16.4+)'));
   310→    }
   311→    var st = new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'));
   312→    return new Response(st).blob().then(function(b){
   313→      return URL.createObjectURL(new Blob([b], { type: mime }));
   314→    });
   315→  }
   316→  return Promise.resolve(URL.createObjectURL(new Blob([bytes], { type: mime })));
