   523→function applyBackground(){
   524→  var bgKey = localStorage.getItem(LS.BG) || 'default';
   525→  var customUrl = localStorage.getItem(LS.BG_CUSTOM) || '';
   526→  var opacity = localStorage.getItem(LS.BG_OPACITY);
   527→  if(opacity === null) opacity = 55;
   528→  var blur = localStorage.getItem(LS.BG_BLUR);
   529→  if(blur === null) blur = 'true';
   530→
   531→  var bgImage = $('bgImage');
   532→  var bgOverlay = $('bgOverlay');
   533→
   534→  var url = '';
   535→  if(bgKey === 'custom' && customUrl){
   536→    url = customUrl;
   537→  } else if(BG_PRESETS[bgKey]){
   538→    url = BG_PRESETS[bgKey];
   539→  } else if(bgKey === 'none'){
   540→    url = '';
   541→  } else {
   542→    url = BG_PRESETS.default;
   543→  }
   544→
   545→  if(url){
   546→    bgImage.style.backgroundImage = 'url(' + url + ')';
   547→    bgImage.style.display = 'block';
   548→  } else {
   549→    bgImage.style.backgroundImage = 'none';
   550→    bgImage.style.display = 'none';
   551→  }
   552→
   553→  bgOverlay.style.background = 'rgba(5,5,5,' + (opacity / 100) + ')';
   554→  if(blur === 'true'){
   555→    bgOverlay.style.backdropFilter = 'blur(4px)';
   556→    bgOverlay.style.webkitBackdropFilter = 'blur(4px)';
   557→  } else {
   558→    bgOverlay.style.backdropFilter = 'none';
   559→    bgOverlay.style.webkitBackdropFilter = 'none';
   560→  }
   561→
   562→  var opts = document.querySelectorAll('.bg-option');
   563→  for(var i = 0; i < opts.length; i++){
   564→    opts[i].classList.remove('active');
   565→    if(opts[i].dataset.bg === bgKey) opts[i].classList.add('active');
   566→  }
   567→
   568→  if(bgKey === 'custom' && customUrl){
   569→    $('bgCustomPreview').style.backgroundImage = 'url(' + customUrl + ')';
   570→  }
   571→
   572→  $('overlayRange').value = opacity;
   573→  $('overlayValue').textContent = opacity + '%';
   574→
   575→  var blurToggle = $('blurToggle');
   576→  if(blur === 'true') blurToggle.classList.add('on');
   577→  else blurToggle.classList.remove('on');
   578→}

   795→function Particle(){
   796→  this.x = Math.random() * canvas.width;
   797→  this.y = Math.random() * canvas.height;
   798→  this.vx = (Math.random() - 0.5) * 0.4;
   799→  this.vy = (Math.random() - 0.5) * 0.4;
   800→  this.size = Math.random() * 2 + 0.5;
   801→  this.opacity = Math.random() * 0.4 + 0.1;
   802→}
   803→Particle.prototype.update = function(){
   804→  this.x += this.vx; this.y += this.vy;
   805→  var dx = mouseX - this.x, dy = mouseY - this.y;
   806→  var dist = Math.sqrt(dx*dx + dy*dy);
   807→  if(dist < 150){ this.vx -= dx * 0.0003; this.vy -= dy * 0.0003; }
   808→  if(this.x < 0 || this.x > canvas.width) this.vx *= -1;
   809→  if(this.y < 0 || this.y > canvas.height) this.vy *= -1;
   810→};
   811→Particle.prototype.draw = function(){
   812→  ctx.beginPath();
   813→  ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
   814→  ctx.fillStyle = 'rgba(245,158,11,' + this.opacity + ')';
   815→  ctx.fill();
   816→};

   818→function animateParticles(){
   819→  if(!particlesEnabled || !ctx) return;
   820→  ctx.clearRect(0, 0, canvas.width, canvas.height);
   821→  for(var i = 0; i < particles.length; i++){
   822→    for(var j = i + 1; j < particles.length; j++){
   823→      var dx = particles[i].x - particles[j].x;
   824→      var dy = particles[i].y - particles[j].y;
   825→      var dist = Math.sqrt(dx*dx + dy*dy);
   826→      if(dist < 120){
   827→        ctx.beginPath();
   828→        ctx.moveTo(particles[i].x, particles[i].y);
   829→        ctx.lineTo(particles[j].x, particles[j].y);
   830→        ctx.strokeStyle = 'rgba(245,158,11,' + (0.08 * (1 - dist/120)) + ')';
   831→        ctx.lineWidth = 0.5;
   832→        ctx.stroke();
   833→      }
   834→    }
   835→  }
   836→  for(var k = 0; k < particles.length; k++){
   837→    particles[k].update();
   838→    particles[k].draw();
   839→  }
   840→  requestAnimationFrame(animateParticles);
   841→}
