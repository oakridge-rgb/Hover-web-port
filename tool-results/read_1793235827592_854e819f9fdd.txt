   579→   579→    } catch(err){
   580→   580→      if (seq !== FATE_PLAY_SEQ) return;
   581→   581→      /* host refuses embedding AND code download -> hard error; otherwise fall through to direct embed */
   582→   582→      if (FATE_NO_EMBED.some(rx => rx.test(fateHost(g.url))) && !g.frameFallback){

   586→   586→        fateBumpRecent(g);
   587→   587→        return;
   588→   588→      }
   589→   589→      fatePStatus('code fetch failed \u2014 trying direct embed\u2026');
   590→   590→    }
   591→   591→  } else {
   592→   592→    fateShowSpin(true, 'Loading ' + g.name + '\u2026', g.source + ' \u00b7 direct CDN embed');
   593→   593→  }
   594→   594→  /* direct embed */
   595→   595→  if (seq !== FATE_PLAY_SEQ) return;

   597→   597→  frame.src = g.url;
   598→   598→  fatePStatus('direct CDN embed \u00b7 black screen? tap \u24e3 Code', '');
   599→   599→  fateArmHint(seq);
   600→   600→  fateBumpRecent(g);
